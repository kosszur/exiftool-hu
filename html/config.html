#------------------------------------------------------------------------------
# Fájl:         example.config  -->  ~/.ExifTool_config
#
# Leírás:       Példa felhasználói konfigurációs fájl az Image::ExifTool-hoz
#
# Megjegyzések: Ez a példafájl bemutatja, hogyan definiálhatsz saját
#               rövidítéseket, és hogyan adhatsz hozzá új EXIF, IPTC, XMP, PNG,
#               MIE és Composite tageket, valamint hogyan adhatod meg a
#               preferált objektíveket a LensID taghez, és hogyan definiálhatsz
#               új fájltípusokat, valamint alapértelmezett ExifTool API-opciókat
#               és parancssori argumentumokat.
#
#               Vedd figyelembe, hogy az ismeretlen tagek akkor is kinyerhetők,
#               ha nincsenek definiálva, de a tageket definiálni kell ahhoz,
#               hogy írhatók legyenek.  Vedd figyelembe azt is, hogy egy meglévő
#               tagdefiníció felülbírálható egy felhasználó által definiált
#               taggel.
#
#               A fájl aktiválásához nevezd át ".ExifTool_config"-ra, és
#               helyezd a home könyvtáradba vagy az exiftool alkalmazás
#               könyvtárába.  (Macen és néhány Windows-rendszeren ezt a
#               parancssorból kell megtenni, mert a grafikus felület esetleg
#               nem engedi, hogy a fájlnevek ponttal kezdődjenek.  Használd a
#               "rename" parancsot Windowson, vagy az "mv"-t Macen.)  Ennek
#               hatására az ExifTool automatikusan betölti a fájlt futtatáskor.
#               A home könyvtáradat a következő környezeti változók közül az
#               első definiált határozza meg:
#
#                   1.  EXIFTOOL_HOME
#                   2.  HOME
#                   3.  HOMEDRIVE + HOMEPATH
#                   4.  (az aktuális könyvtár)
#
#               Alternatívaként az exiftool alkalmazás -config opciója
#               használható egy adott konfigurációs fájl betöltésére (vedd
#               figyelembe, hogy ennek kell lennie a parancssor első opciójának):
#
#                   exiftool -config example.config ...
#
#               Ez a példafájl a következő 16 új taget definiálja, valamint
#               néhány Shortcut és Composite taget:
#
#                   1.  EXIF:NewEXIFTag
#                   2.  GPS:GPSPitch
#                   3.  GPS:GPSRoll
#                   4.  IPTC:NewIPTCTag
#                   5.  XMP-xmp:NewXMPxmpTag
#                   6.  XMP-exif:GPSPitch
#                   7.  XMP-exif:GPSRoll
#                   8.  XMP-xxx:NewXMPxxxTag1
#                   9.  XMP-xxx:NewXMPxxxTag2
#                  10.  XMP-xxx:NewXMPxxxTag3
#                  11.  XMP-xxx:NewXMPxxxStruct
#                  12.  PNG:NewPngTag1
#                  13.  PNG:NewPngTag2
#                  14.  PNG:NewPngTag3
#                  15.  MIE-Meta:NewMieTag1
#                  16.  MIE-Test:NewMieTag2
#
#               A tag-táblák és a taginformációs hashek definíciójáról részletes
#               információért lásd a lib/Image/ExifTool/README fájlt.
#------------------------------------------------------------------------------

# A Shortcut (rövidítés) tageket információ kinyerésekor használjuk a

# jelölésére, vagy egy tagnév aliasának megadására.
%Image::ExifTool::UserDefined::Shortcuts = (
    MyShortcut => ['exif:createdate','exposuretime','aperture'],
    MyAlias => 'FocalLengthIn35mmFormat',
);

# MEGJEGYZÉS: A következő táblákban használt összes tagnév kis-/nagybetűérzékeny.

# A %Image::ExifTool::UserDefined hash a meglévő táblákhoz hozzáadandó
# új tageket definiálja.
%Image::ExifTool::UserDefined = (
    # A GPS-en kívül minden EXIF tag a Main táblához adódik hozzá, és a
    # WriteGroup adja meg, hová íródik a tag (alapértelmezésben ExifIFD).
    # Vedd figyelembe, hogy egyedi EXIF tagek olvasásakor ugyanazt a
    # konfigurációs fájlt kell használni, hogy a megadott nevekkel láthatók
    # legyenek. A konfigurációs fájl nélkül csak ismeretlen tagekként
    # láthatók a -u opcióval.
    'Image::ExifTool::Exif::Main' => {
        # 1. példa.  EXIF:NewEXIFTag
        0xd000 => {
            Name => 'NewEXIFTag',
            Writable => 'int16u',
            WriteGroup => 'IFD0',
        },
        # ide vegyél fel további, felhasználó által definiált EXIF tageket...
    },
    # a Geotag szolgáltatás GPSPitch és GPSRoll tageket ír, de ezek
    # nem szabványos EXIF-ek, így itt egyedi tageket definiálunk:
    'Image::ExifTool::GPS::Main' => {
        # 2. példa.  GPS:GPSPitch
        0xd000 => {
            Name => 'GPSPitch',
            Writable => 'rational64s',
        },
        # 3. példa.  GPS:GPSRoll
        0xd001 => {
            Name => 'GPSRoll',
            Writable => 'rational64s',
        },
    },
    # Az IPTC tagek egy adott rekordtípushoz adódnak hozzá (pl. application record):
    # (Megjegyzés: az IPTC tag ID-k a 0-255 tartományra korlátozódnak)
    'Image::ExifTool::IPTC::ApplicationRecord' => {
        # 4. példa.  IPTC:NewIPTCTag
        160 => {
            Name => 'NewIPTCTag',
            Format => 'string[0,16]',
        },
        # ide vegyél fel további, felhasználó által definiált IPTC ApplicationRecord tageket...
    },
    # XMP tagek hozzáadhatók meglévő névterekhez:
    'Image::ExifTool::XMP::xmp' => {
        # 5. példa.  XMP-xmp:NewXMPxmpTag
        NewXMPxmpTag => { Groups => { 2 => 'Author' } },
        # ide vegyél fel további, felhasználó által definiált XMP-xmp tageket...
    },
    # speciális Geotag tagek az XMP-exif-hez:
    'Image::ExifTool::XMP::exif' => {
        # 6. példa.  XMP-exif:GPSPitch
        GPSPitch => { Writable => 'rational', Groups => { 2 => 'Location' } },
        # 7. példa.  XMP-exif:GPSRoll
        GPSRoll  => { Writable => 'rational', Groups => { 2 => 'Location' } },
    },
    # az új XMP-névtereket (pl. xxx) a Main XMP táblához kell hozzáadni:
    'Image::ExifTool::XMP::Main' => {
        # névtér-definíció a 8-11. példához
        xxx => { # <-- meg kell egyeznie a NAMESPACE előtaggal
            SubDirectory => {
                TagTable => 'Image::ExifTool::UserDefined::xxx',
                # (lásd ennek a táblának a definícióját lentebb)
            },
        },
        # ide vegyél fel további, felhasználó által definiált XMP-névtereket...
    },
    # az új PNG tagek a PNG::TextualData táblához adódnak hozzá:
    'Image::ExifTool::PNG::TextualData' => {
        # 12. példa.  PNG:NewPngTag1
        NewPngTag1 => { }, # (a Writable alapértelmezésben 'string' az XMP tageknél)
        # 13. példa.  PNG:NewPngTag2
        NewPngTag2 => { iTXt => 1 }, # (kényszeríti, hogy ez a tag iTXt-ként íródjon)
        # 14. példa.  PNG:NewPngTag3
        NewPngTag3 => { },
    },
    # új MIE tag (NewMieTag1) és csoport (MIE-Test) hozzáadása a MIE-Meta-hoz
    # (Megjegyzés: a MIE csoportnevek NEM végződhetnek számmal)
    'Image::ExifTool::MIE::Meta' => {
        # 15. példa.  MIE-Meta:NewMieTag1
        NewMieTag1 => {
            Writable => 'rational64u',
            Units => [ 'cm', 'in' ],
        },
        # új MIE „Test" csoport a 16. példához
        Test => {
            SubDirectory => {
                TagTable => 'Image::ExifTool::UserDefined::MIETest',
                DirName => 'MIE-Test',
            },
        },
    },
    # a Composite tagek a Composite táblához adódnak hozzá:
    'Image::ExifTool::Composite' => {
        # a Composite tagek értékei más tagek értékeiből származnak.  Az
        # alábbi Require/Desire elemek olyan alkotórész-tageket adnak meg,
        # amelyeknek léteznie kell / létezhet, és ezeknek a hasheknek a
        # kulcsai indexként szolgálnak a ValueConv kifejezés @val tömbjében
        # ezen tagek numerikus (-n) értékeinek eléréséhez.  A
        # kiírás-konvertált értékek a @prt tömbön keresztül érhetők el.  Az
        # összes Require-elt tagnek léteznie kell ahhoz, hogy a Composite tag
        # kiértékelődjön.  Ha nincs Require-elt tag megadva, akkor a
        # Desire-elt tagek közül legalább egynek léteznie kell.  További
        # példákért lásd a Composite táblát az Image::ExifTool::Exif-ben, az
        # összes részletért pedig a lib/Image/ExifTool/README fájlt.
        # Az első néhány példa olyan egyszerűsítéseket mutat be, amelyek
        # akkor használhatók, ha csak egy tag Require-elt vagy Desire-elt:
        # 1) a Require kikeresés helyettesíthető egy egyszerű tagnévvel
        # 2) a "$val" használható a "$val[0]" jelölésére a kifejezésben
        FileExtension => {
            Require => 'FileName',
            ValueConv => '$val=~/\.([^.]*)$/; $1',
        },
        # a CircleOfConfusion tag felülbírálása, hogy D/1750-et használjon D/1440 helyett
        CircleOfConfusion => {
            Require => 'ScaleFactor35efl',
            Groups => { 2 => 'Camera' },
            ValueConv => 'sqrt(24*24+36*36) / ($val * 1750)',
            # egy opcionális PrintConv használható az érték formázására
            PrintConv => 'sprintf("%.3f mm",$val)',
        },
        # leírás generálása ehhez a fájltípushoz
        FileTypeDescription => {
            Require => 'FileType',
            ValueConv => 'GetFileType($val,1) || $val',
        },
        # a fizikai képméret kiszámítása a felbontás alapján
        PhysicalImageSize => {
            Require => {
                0 => 'ImageWidth',
                1 => 'ImageHeight',
                2 => 'XResolution',
                3 => 'YResolution',
                4 => 'ResolutionUnit',
            },
            ValueConv => '$val[0]/$val[2] . " " . $val[1]/$val[3]',
            # (a @prt tömb kiírásra formázott értékeket tartalmaz)
            PrintConv => 'sprintf("%.1fx%.1f $prt[4]", split(" ",$val))',
        },
        # [haladó] a legnagyobb JPEG előnézeti kép kiválasztása
        BigImage => {
            Groups => { 2 => 'Preview' },
            Desire => {
                0 => 'JpgFromRaw',
                1 => 'PreviewImage',
                2 => 'OtherImage',
                # (a DNG-nek és az A100 ARW-nak lehet 2 PreviewImage-e)
                3 => 'PreviewImage (1)',
                # (ha az MPF-nek 2 előnézete van, az MPImage3 lehet a nagyobbik)
                4 => 'MPImage3',
                5 => 'JpgFromRaw2', # újabb Panasonic modellek RW2-jében
            },
            # a ValueConv kódreferencia is lehet
            # Bemenetek: 0) hivatkozás az értékek listájára, 1) ExifTool objektum
            ValueConv => sub {
                my $val = shift;
                my ($image, $bigImage, $len, $bigLen);
                foreach $image (@$val) {
                    next unless ref $image eq 'SCALAR';
                    # JPEG kép ellenőrzése (vagy „Binary data", ha nincs -b használva)
                    next unless $$image =~ /^(\xff\xd8\xff|Binary data (\d+))/;
                    $len = $2 || length $$image; # a kép hosszának lekérése
                    # a legnagyobb kép mentése
                    next if defined $bigLen and $bigLen >= $len;
                    $bigLen = $len;
                    $bigImage = $image;
                }
                return $bigImage;
            },
        },
        # **** IDE VEGYÉL FEL TOVÁBBI COMPOSITE TAGDEFINÍCIÓKAT ****
    },
);

# Ez egy alap példa egy új XMP-névtér definíciójára.
# Erre a táblára a fenti %Image::ExifTool::UserDefined definícióban lévő
# SubDirectory tagdefiníción keresztül hivatkozunk.
# Ezeknek a tageknek a névtér-előtagja 'xxx', ami az ExifTool 'XMP-xxx'
# nevű 1. családbeli csoportnevének felel meg.
%Image::ExifTool::UserDefined::xxx = (
    GROUPS => { 0 => 'XMP', 1 => 'XMP-xxx', 2 => 'Image' },
    NAMESPACE => { 'xxx' => 'http://ns.myname.com/xxx/1.0/' },
    WRITABLE => 'string', # (alapértelmezésben string típusú tagek)
    # 8. példa.  XMP-xxx:NewXMPxxxTag1 (más nyelvi tag)
    # - cseréld a "NewXMPxxxTag1"-et a saját tagnevedre (pl. "MyTag")
    NewXMPxxxTag1 => { Writable => 'lang-alt' },
    # 9. példa.  XMP-xxx:NewXMPxxxTag2 (string tag az Author kategóriában)
    NewXMPxxxTag2 => { Groups => { 2 => 'Author' } },
    # 10. példa.  XMP-xxx:NewXMPxxxTag3 (rendezetlen List típusú tag)
    NewXMPxxxTag3 => { List => 'Bag' },
    # 11. példa.  XMP-xxx:NewXMPxxxStruct (strukturált tag)
    # - példa strukturált XMP tagre
    NewXMPxxxStruct => {
        # a "Struct" bejegyzés definiálja a struktúra mezőit
        Struct => {
            # opcionális névtér-előtag és URI a struktúramezőkhöz
            # (csak akkor kötelező, ha eltér a fenti NAMESPACE-től)
            # --> ebben a névtér-kikeresésben több bejegyzés is lehet,
            # ahol az ábécében utolsó az alapértelmezett a mezőkhöz, de
            # minden mezőnek lehet "Namespace" eleme, amely megadja,
            # melyik előtagot használja
            NAMESPACE => { 'test' => 'http://x.y.z/test/' },
            # opcionális struktúranév (csak figyelmeztető üzenetekhez használatos)
            STRUCT_NAME => 'MyStruct',
            # opcionális rdf:type property a struktúrához
            TYPE => 'http://x.y.z/test/xystruct',
            # struktúramező-definíciók (nagyon hasonlóak a tagdefiníciókhoz)
            X => { Writable => 'integer' },
            Y => { Writable => 'integer' },
            # egy beágyazott struktúra...
            Things => {
                List => 'Bag',
                Struct => {
                    NAMESPACE => { thing => 'http://x.y.z/thing/' },
                    What  => { },
                    Where => { },
                },
            },
        },
        List => 'Seq', # a struktúrák egy lista elemei is lehetnek
    },
    # A struktúra minden mezőjéhez tartozik egy megfelelő lapított tag, amely
    # automatikusan generálódik, ID-je az eredeti struktúra-tag ID és a
    # mezőnév összefűzéséből áll (a mezőnév első betűjét szükség esetén
    # nagybetűsítve).  Ezeknek a lapított tageknek a Name és/vagy Description
    # értéke megváltoztatható, ha kívánod, de minden más tag-tulajdonság a
    # struktúramező-definícióból származik.  A "Flat" jelzőt kell használni egy
    # lapított tag Name vagy Description értékének beállításakor.  Például:
    NewXMPxxxStructX => { Name => 'SomeOtherName', Flat => 1 },
);

# Felhasználó által definiált, felismerendő fájltípusok
%Image::ExifTool::UserDefined::FileTypes = (
    XXX => { # <-- az új fájltípus kiterjesztése (kis-/nagybetűre nem érzékeny)
        # a BaseType adja meg a formátumot, amelyen ez a fájl alapul
        # (kis-/nagybetűérzékeny).  Ha a BaseType definiálva van, akkor a
        # fájl teljesen támogatott lesz, és ebben az esetben a Magic mintát
        # nem szabad definiálni
        BaseType => 'TIFF',
        MIMEType => 'image/x-xxx',
        Description => 'My XXX file type',
        # ha a BaseType írható az ExifToollal, akkor az új fájltípus is
        # írható lesz, hacsak másképp nincs megadva, így:
        Writable => 0,
    },
    YYY => {
        # BaseType nélkül a fájlt felismeri, de nem támogatja
        Magic => '0123abcd',    # a fájl elején illesztendő reguláris kifejezés
        MIMEType => 'application/test',
        Description => 'My YYY file type',
    },
    ZZZ => {
        # ha sem a BaseType, sem a Magic nincs definiálva, a fájlt csak
        # kiterjesztés alapján ismeri fel.  A MIMEType application/unknown
        # lesz, hacsak másképp nincs megadva
        Description => 'My ZZZ file type',
    },
    # ha csak a BaseType van megadva, akkor a következő egyszerűsített
    # szintaxis használható.  Ebben a példában a "TTT" kiterjesztésű fájlok
    # JPEG fájlként kerülnek feldolgozásra
    TTT => 'JPEG',
);
